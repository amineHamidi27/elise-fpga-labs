% State diagram of fsm_spi (Moore machine). Build: pdflatex state_diagram.tex
\documentclass[border=6pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{automata,positioning,arrows.meta}
\begin{document}
\begin{tikzpicture}[
    >=Stealth, auto, node distance=3.2cm and 6cm, semithick,
    every state/.style={draw, circle, minimum size=3.4cm, inner sep=1pt, align=center, font=\small},
    lbl/.style={font=\small}]

  \node[state, initial, initial text=reset] (idle) {\textbf{IDLE}\\[2pt] fsync = 1\\ enable = 0};
  \node[state, right=of idle]  (start) {\textbf{START\_SYNC}\\[2pt] fsync = 0\\ enable = 0};
  \node[state, below=of start] (send)  {\textbf{SENDING}\\[2pt] fsync = 0\\ enable = 1};
  \node[state, below=of idle]  (wait)  {\textbf{WAIT\_RELEASE}\\[2pt] fsync = 1\\ enable = 0};

  \path[->]
    (idle)  edge[lbl]                     node {button = 1} (start)
    (start) edge[lbl]                     node {1}          (send)
    (send)  edge[lbl]                     node {fin = 1}    (wait)
    (wait)  edge[lbl]                     node {button = 0} (idle)
    (idle)  edge[loop above, lbl]         node {button = 0} ()
    (send)  edge[loop right, lbl]         node {fin = 0}    ()
    (wait)  edge[loop left, lbl]          node {button = 1} ();
\end{tikzpicture}
\end{document}
